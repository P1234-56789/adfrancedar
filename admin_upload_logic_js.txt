/**
 * Admin Upload Logic for ADFRAN Cedar
 * Integrates with firebase.js utility and Firestore/Storage.
 */

import { storage, db, auth } from './firebase';
import { ref, uploadBytesResumable, getDownloadURL } from "firebase/storage";
import { collection, addDoc, serverTimestamp } from "firebase/firestore";

/**
 * Handles the "Post to Catalog" action.
 * @param {File} file - The image file from the input/camera.
 * @param {Object} metadata - { title, description, category }
 * @param {Function} onProgress - Callback for upload percentage.
 */
export const handlePostToCatalog = async (file, metadata, onProgress) => {
  const user = auth.currentUser;
  
  // 1. Verify Admin Authentication
  if (!user || user.email !== 'francis@adfrancedar.com') {
    throw new Error("Unauthorized: Only Master Craftsman Francis can post to the catalog.");
  }

  // 2. Prepare Storage Reference
  const storageRef = ref(storage, `catalog/${Date.now()}_${file.name}`);
  const uploadTask = uploadBytesResumable(storageRef, file);

  return new Promise((resolve, reject) => {
    uploadTask.on('state_changed', 
      (snapshot) => {
        const progress = (snapshot.bytesTransferred / snapshot.totalBytes) * 100;
        if (onProgress) onProgress(progress);
      }, 
      (error) => {
        console.error("Upload failed:", error);
        reject(error);
      }, 
      async () => {
        // 3. Get Public URL
        const downloadURL = await getDownloadURL(uploadTask.snapshot.ref);

        // 4. Sync to Firestore
        try {
          const docRef = await addDoc(collection(db, "furniture_catalog"), {
            title: metadata.title,
            description: metadata.description,
            category: metadata.category,
            imageUrl: downloadURL,
            storagePath: storageRef.fullPath,
            createdAt: serverTimestamp(),
            createdBy: user.email,
            status: 'published'
          });
          
          resolve({ id: docRef.id, url: downloadURL });
        } catch (dbError) {
          console.error("Firestore sync failed:", dbError);
          reject(dbError);
        }
      }
    );
  });
};
